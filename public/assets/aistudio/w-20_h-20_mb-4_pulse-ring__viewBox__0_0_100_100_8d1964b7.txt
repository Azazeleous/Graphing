    // Decode Base64 metadata directly from #cryptic-logo SVG
    function getDecodedIdeology() {
      const logo = document.getElementById('cryptic-logo');
      const rawB64 = logo.getAttribute('data-ideology');
      try {
        return atob(rawB64);
      } catch (e) {
        return "Ideology decode error.";
      }
    }

    function renderIdeology() {
      const decoded = getDecodedIdeology();
      const listEl = document.getElementById('ideology-list');
      // Split by numbered items "1. ", "2. ", etc.
      const parts = decoded.split(/(?=\d+\.\s)/).filter(Boolean);
      listEl.innerHTML = parts.map(item => {
        const [title, ...rest] = item.split(':');
        return `<li class="border-l-2 border-yellow-400/70 pl-2.5 py-0.5">
                  <span class="text-yellow-300 font-semibold">${title}:</span>
                  <span class="text-slate-300">${rest.join(':')}</span>
                </li>`;
      }).join('');
    }

    // Hamiltonian Symplectic Phase-Space Visualizer
    let animStarted = false;
    function initPhaseCanvas() {
      if (animStarted) return;
      animStarted = true;

      const canvas = document.getElementById('phase-canvas');
      const container = document.getElementById('canvas-container');
      const ctx = canvas.getContext('2d');
      const slider = document.getElementById('coupling-slider');
      const energyReadout = document.getElementById('energy-readout');
      const syncCounter = document.getElementById('sync-counter');
      const hzReadout = document.getElementById('telemetry-hz');

      function resizeCanvas() {
        canvas.width = container.clientWidth;
        canvas.height = container.clientHeight;
      }
      resizeCanvas();
      window.addEventListener('resize', resizeCanvas);

      // Initialize multiple phase-space orbits (q, p)
      const particles = Array.from({ length: 18 }, (_, i) => ({
        q: (i - 8.5) * 0.35,
        p: ((i % 3) - 1) * 0.6,
        hue: i % 2 === 0 ? '#facc15' : '#22d3ee',
        trail: []
      }));

      let tick = 0;
      const dt = 0.035;

      function step() {
        tick++;
        const kappa = parseFloat(slider.value);
        const w = canvas.width;
        const h = canvas.height;
        const cx = w / 2;
        const cy = h / 2;
        const scale = Math.min(w, h) * 0.16;

        // Subtle fade trail
        ctx.fillStyle = 'rgba(5, 7, 13, 0.18)';
        ctx.fillRect(0, 0, w, h);

        // Draw coordinate axes
        ctx.strokeStyle = 'rgba(148, 163, 184, 0.15)';
        ctx.lineWidth = 1;
        ctx.beginPath();
        ctx.moveTo(0, cy); ctx.lineTo(w, cy);
        ctx.moveTo(cx, 0); ctx.lineTo(cx, h);
        ctx.stroke();

        let sampleEnergy = 0;

        particles.forEach((pt, idx) => {
          // Symplectic Euler integration for H = 0.5*p^2 + 0.5*q^2 + kappa*cos(q)
          const force = -pt.q + kappa * Math.sin(pt.q);
          pt.p += force * dt;
          pt.q += pt.p * dt;

          // Wrap q smoothly for visual continuity
          if (pt.q > 4.5) pt.q -= 9;
          if (pt.q < -4.5) pt.q += 9;

          const sx = cx + pt.q * scale;
          const sy = cy - pt.p * scale;

          ctx.fillStyle = pt.hue;
          ctx.beginPath();
          ctx.arc(sx, sy, 2.2, 0, Math.PI * 2);
          ctx.fill();

          if (idx === 4) {
            sampleEnergy = 0.5 * pt.p * pt.p + 0.5 * pt.q * pt.q + kappa * Math.cos(pt.q);
          }
        });

        if (tick % 15 === 0) {
          energyReadout.textContent = `Invariant Energy E: ${sampleEnergy.toFixed(4)} J`;
          syncCounter.textContent = `SYNC #${String(tick).padStart(4, '0')}`;
          hzReadout.textContent = `${(59.95 + Math.random() * 0.1).toFixed(2)} Hz`;
        }

        requestAnimationFrame(step);
      }

      step();
    }
