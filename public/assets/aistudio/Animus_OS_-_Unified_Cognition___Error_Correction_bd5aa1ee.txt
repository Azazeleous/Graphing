function encodeULEB128(value) {
    if (!Number.isSafeInteger(value) || value < 0) {
        throw new RangeError("ULEB128 parameters require non-negative safe integers.");
    }
    const outputBytes = [];
    do {
        let singleByte = value % 128;
        value = Math.floor(value / 128);
        if (value !== 0) {
            singleByte |= 0x80;
        }
        outputBytes.push(singleByte);
    } while (value !== 0);
    return Uint8Array.from(outputBytes);
}

function decodeULEB128(bytes, offset = 0) {
    let accumulatedValue = 0;
    let weightMultiplier = 1;

    let readPosition = offset;
    while (readPosition < bytes.length) {
        const extractedByte = bytes[readPosition++];
        accumulatedValue += (extractedByte & 0x7f) * weightMultiplier;
        if (!Number.isSafeInteger(accumulatedValue)) {
            throw new RangeError("Decoded lexicon identifier value overflowed safe integer bounds.");
        }
        if ((extractedByte & 0x80) === 0) {
            return { value: accumulatedValue, nextOffset: readPosition };
        }
        weightMultiplier *= 128;
        if (readPosition - offset > 8) {
            throw new RangeError("ULEB128 byte sequence size specification limit exceeded.");
        }
    }
    throw new Error("Incomplete ULEB128 byte sequence tracking parameters.");
}

/* =============================================================================
   MATHEMATICAL SYSTEM LOGIC: FOUR-LANE INVERSE TRACE PLATFORM
   ============================================================================= */
class FourLaneInverseLatticeEngine {
    constructor() {
        this.nodes = [];
        this.elapsedTimeoutTime = 0;
        this.initializeTopology();
    }

    initializeTopology() {
        this.nodes = [
            {x: 1, y: 1, z: 0}, {x: 1, y: -1, z: 0}, {x: -1, y: 1, z: 0}, {x: -1, y: -1, z: 0},
            {x: 1, y: 0, z: 1}, {x: 1, y: 0, z: -1}, {x: -1, y: 0, z: 1}, {x: -1, y: 0, z: -1},
            {x: 0, y: 1, z: 1}, {x: 0, y: 1, z: -1}, {x: 0, y: -1, z: 1}, {x: 0, y: -1, z: -1}
        ];
    }

    calculateHyperSpatialProjection(stepIndex, globalFallbackActive) {

        const fractionalTimingIncrement = (stepIndex % 10) * 0.1;
        const wholeCycleIteration = Math.floor(stepIndex / 10);
        
        let targetYAxis = Math.sin(fractionalTimingIncrement * Math.PI); 
        if (fractionalTimingIncrement === 0.5) {
            targetYAxis = 1.0;
        }

        const leftRightFactor = fractionalTimingIncrement < 0.5 ? -1.0 : 1.0;
        const forwardBackwardZAxis = fractionalTimingIncrement > 0.5 ? 1.5 : -1.5;

        const lane4Forward = wholeCycleIteration % 12;
        const lane3Backward = (12 - 1 - lane4Forward) % 12;

        const baseGaussianCurve = Math.exp(-0.5 * Math.pow(Math.abs((stepIndex % 100) - 50) / 15, 8));
        let shadowTargetCurve = Math.exp(-0.5 * Math.pow(Math.abs(((stepIndex + 25) % 100) - 50) / 15, 8));

        if (globalFallbackActive) {
            this.elapsedTimeoutTime += 0.1;

            shadowTargetCurve += Math.abs(Math.sin(this.elapsedTimeoutTime)) * 0.45;
        }

        return {
            lane1Node: lane4Forward,
            lane3Node: lane3Backward,
            computedY: targetYAxis * leftRightFactor,
            computedZ: forwardBackwardZAxis,
            computedW: Math.cos(fractionalTimingIncrement * Math.PI),
            autoWeight: baseGaussianCurve,
            driftDelta: Math.abs(baseGaussianCurve - shadowTargetCurve) * 0.1
        };
    }
}

let renderingScene, trackingCamera, structuralRenderer;

let pipelineProcessingActive = true;
let cycleTicker = 0;
let globalInversionStateActive = false;
const mathEngineInstance = new FourLaneInverseLatticeEngine();
window.activeColorHueShiftIntensity = 0.0;

function initSystemVisualizer() {
    const renderBoxContainer = document.getElementById('hologram-viewport');
    const w = renderBoxContainer.clientWidth;
    const h = renderBoxContainer.clientHeight;

    renderingScene = new THREE.Scene();
    trackingCamera = new THREE.PerspectiveCamera(45, w / h, 0.1, 100);
    trackingCamera.position.set(0, 0, 7);

    structuralRenderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
    structuralRenderer.setSize(w, h);
    renderBoxContainer.appendChild(structuralRenderer.domElement);


    const modelGeometryBuffer = new THREE.BufferGeometry();
    const linearPointsArray = [];
    
    mathEngineInstance.nodes.forEach(node => {
        linearPointsArray.push(node.x * 1.6, node.y * 1.6, node.z * 1.6);
    });

    modelGeometryBuffer.setAttribute('position', new THREE.Float32BufferAttribute(linearPointsArray, 3));

    const elementIndicesMapping = [];
    for(let outer=0; outer < mathEngineInstance.nodes.length; outer++) {
        for(let inner=outer+1; inner < mathEngineInstance.nodes.length; inner++) {
            const dx = mathEngineInstance.nodes[outer].x - mathEngineInstance.nodes[inner].x;
            const dy = mathEngineInstance.nodes[outer].y - mathEngineInstance.nodes[inner].y;
            const dz = mathEngineInstance.nodes[outer].z - mathEngineInstance.nodes[inner].z;
            const distanceSquared = dx*dx + dy*dy + dz*dz;
            if(Math.abs(distanceSquared - 2.0) < 0.1) {
                elementIndicesMapping.push(outer, inner);
            }

        }
    }
    modelGeometryBuffer.setIndex(elementIndicesMapping);

    const wireframeStyleMaterial = new THREE.MeshBasicMaterial({
        color: 0x42eaff,
        wireframe: true,
        transparent: true,
        opacity: 0.35
    });

    window.crystallineLatticeWireframe = new THREE.LineSegments(modelGeometryBuffer, wireframeStyleMaterial);
    renderingScene.add(window.crystallineLatticeWireframe);

    const markerGeometry = new THREE.SphereGeometry(0.12, 16, 16);
    const autoTrackMaterial = new THREE.MeshBasicMaterial({ color: 0x42eaff });
    const shadowTrackMaterial = new THREE.MeshBasicMaterial({ color: 0xef4444 });

    window.lane1MarkerMesh = new THREE.Mesh(markerGeometry, autoTrackMaterial);

    window.lane3MarkerMesh = new THREE.Mesh(markerGeometry, shadowTrackMaterial);
    renderingScene.add(window.lane1MarkerMesh);
    renderingScene.add(window.lane3MarkerMesh);

    executeSystemRenderLoop();
}

function executeSystemRenderLoop() {
    requestAnimationFrame(executeSystemRenderLoop);
    
    if (pipelineProcessingActive) {
        cycleTicker++;
        
        const traversalState = mathEngineInstance.calculateHyperSpatialProjection(cycleTicker, globalInversionStateActive);
        document.getElementById('drift-readout').textContent = traversalState.driftDelta.toFixed(6) + " ms";

        const nodeRef1 = mathEngineInstance.nodes[traversalState.lane1Node % 12];
        const nodeRef3 = mathEngineInstance.nodes[traversalState.lane3Node % 12];


        window.lane1MarkerMesh.position.set(nodeRef1.x * 1.6 + traversalState.computedY, nodeRef1.y * 1.6, nodeRef1.z * 1.6);
        window.lane3MarkerMesh.position.set(nodeRef3.x * 1.6, nodeRef3.y * 1.6 + traversalState.computedZ * 0.1, nodeRef3.z * 1.6 + traversalState.computedW * 0.2);

        const targetPulseScale = 1.0 + traversalState.autoWeight * 0.22;
        window.crystallineLatticeWireframe.scale.set(targetPulseScale, targetPulseScale, targetPulseScale);

        if (window.activeColorHueShiftIntensity > 0.01) {
            const structuralHueCycleValue = (cycleTicker * 2) % 360;
            window.crystallineLatticeWireframe.material.color.setHSL(structuralHueCycleValue / 360, 1.0, 0.6);
            window.activeColorHueShiftIntensity *= 0.98;
        } else {
            window.crystallineLatticeWireframe.material.color.setHex(0x42eaff);
        }

        window.crystallineLatticeWireframe.rotation.x += globalInversionStateActive ? 0.015 : 0.003;
        window.crystallineLatticeWireframe.rotation.y += 0.005;
    }

    structuralRenderer.render(renderingScene, trackingCamera);
}
