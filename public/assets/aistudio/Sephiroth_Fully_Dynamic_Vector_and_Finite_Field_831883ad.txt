                            .add(localVertices[combination[0]])
                            .add(localVertices[combination[1]])

                        const rivetColorPhase = (angle * 1.2) + (rivet.zBase * 0.04) - (independentColorWaveFront * globalSpinDirection);
                        const baselineHue = Math.abs(Math.sin(rivetColorPhase)) % 1.0;
                        const finalHue = chaosOverflow(secondsElapsed, rivet.pIndex, baselineHue);

                        const normalizedDeg = (Math.abs(angle) * (180 / Math.PI)) % 360;
                        const ledStateIndex = Math.floor(normalizedDeg / 15);
                        
                        let ledOn = (ledStateIndex % 2 === 0);

                        if (sys.isInverted) ledOn = !ledOn;

                        if (ledOn) {
                            rivet.mesh.material.color.setHSL(finalHue, 0.95, 0.5);
                            rivet.mesh.material.opacity = 0.5;
                        } else {
                            rivet.mesh.material.color.setHSL(finalHue, 0.2, 0.12);
                            rivet.mesh.material.opacity = 0.08;
                        }
                    });
                });

                const projectedPositions = [];
                const wOscillation = Math.sin(secondsElapsed * 1.1 + (sys.isInverted ? Math.PI : 0)) * 0.35;
                const points3D = sys.tesseractVertices4D.map(v => project4DTo3D(v, wOscillation));
                
                for (let i = 0; i < points3D.length; i++) {
                    for (let j = i + 1; j < points3D.length; j++) {
                        let bitDifference = 0;

                        let checkXor = i ^ j;
                        while (checkXor > 0) {
                            if (checkXor & 1) bitDifference++;
                            checkXor >>= 1;
                        }
                        if (bitDifference === 1) {
                            projectedPositions.push(points3D[i].x, points3D[i].y, points3D[i].z);
                            projectedPositions.push(points3D[j].x, points3D[j].y, points3D[j].z);
                        }
                    }
                }
                sys.tessLines.geometry.setAttribute('position', new THREE.Float32BufferAttribute(projectedPositions, 3));
                sys.tessLines.geometry.attributes.position.needsUpdate = true;

                const mappedVertices = points3D.map((v, idx) => ({ v, idx, dist: v.length() }));
                mappedVertices.sort((a, b) => a.dist - b.dist);
                const innerHorizonPoints = mappedVertices.slice(0, 8).map(item => item.v);
                const outerHorizonPoints = mappedVertices.slice(8, 16).map(item => item.v);


                const quadStabilizerPos = [];
                const explicitCornerTetherPos = [];
                const closedDualbellTetherPos = [];
                const horizontalMirrorPos = [];

                const jInvariantTorsionScale = 1.0 + Math.sin(secondsElapsed * 1.4 + (sys.isInverted ? Math.PI : 0)) * 0.22;
                const nodeCountTotal = sys.nodesA.length;

                sys.nodesA.forEach((node) => {
                    const uData = node.userData;
                    const traversalRate = uData.isEven ? 0.32 : -0.32;
                    const currentSpiralAngle = uData.theta + gateAngleTwist + (secondsElapsed * (traversalRate / (uData.index * 0.005 + 1)));

                    const calculatedX = Math.cos(currentSpiralAngle) * uData.rBase * jInvariantTorsionScale;
                    const calculatedY = Math.sin(currentSpiralAngle) * uData.rBase * jInvariantTorsionScale;
                    const calculatedZ = uData.zBase; 

                    uData.history.push(new THREE.Vector3(calculatedX, calculatedY, calculatedZ));
                    if (uData.history.length > 4) uData.history.shift();


                    node.position.copy(uData.history[0]);
                    
                    const colorPhase = (currentSpiralAngle * 1.8) + (calculatedZ * 0.05) - (independentColorWaveFront * globalSpinDirection);
                    const coreHue = chaosOverflow(secondsElapsed, uData.index, Math.abs(Math.sin(colorPhase)) % 1.0);
                    node.material.color.setHSL(coreHue, 0.95, 0.5);

                    const targetVertex = outerHorizonPoints[uData.index % outerHorizonPoints.length];
                    quadStabilizerPos.push(node.position.x, node.position.y, node.position.z);
                    quadStabilizerPos.push(targetVertex.x, targetVertex.y, targetVertex.z);
                });

                sys.nodesB.forEach((node) => {
                    const uData = node.userData;
                    const traversalRate = uData.isEven ? -0.32 : 0.32;
                    
                    const coprimePhaseShift = ((uData.index * currentPrimeSeed) % nodeCountTotal / nodeCountTotal) * Math.PI * 2;
                    const dynamicThetaB = -uData.theta + coprimePhaseShift;
                    const currentSpiralAngle = dynamicThetaB - gateAngleTwist + (secondsElapsed * (traversalRate / (uData.index * 0.005 + 1)));


                    const calculatedX = Math.cos(currentSpiralAngle) * uData.rBase * jInvariantTorsionScale;
                    const calculatedY = Math.sin(currentSpiralAngle) * uData.rBase * jInvariantTorsionScale;
                    const calculatedZ = uData.zBase;

                    uData.history.push(new THREE.Vector3(calculatedX, calculatedY, calculatedZ));
                    if (uData.history.length > 4) uData.history.shift();

                    node.position.copy(uData.history[0]);

                    const colorPhase = (currentSpiralAngle * 1.8) + (calculatedZ * 0.05) - (independentColorWaveFront * globalSpinDirection);
                    const coreHue = chaosOverflow(secondsElapsed, uData.index + 90, Math.abs(Math.sin(colorPhase)) % 1.0);
                    node.material.color.setHSL(coreHue, 0.95, 0.5);

                    const targetVertex = innerHorizonPoints[uData.index % innerHorizonPoints.length];
                    quadStabilizerPos.push(node.position.x, node.position.y, node.position.z);
                    quadStabilizerPos.push(targetVertex.x, targetVertex.y, targetVertex.z);
                });

                innerHorizonPoints.forEach((innerPt, idx) => {

                    const endpointNode = sys.nodesA[idx % Math.max(1, Math.floor(sys.nodesA.length * 0.1))];
                    if (endpointNode) {
                        explicitCornerTetherPos.push(innerPt.x, innerPt.y, innerPt.z);
                        explicitCornerTetherPos.push(endpointNode.position.x, endpointNode.position.y, endpointNode.position.z);
                    }
                });
                outerHorizonPoints.forEach((outerPt, idx) => {
                    const endpointNode = sys.nodesB[sys.nodesB.length - 1 - (idx % Math.max(1, Math.floor(sys.nodesB.length * 0.1)))];
                    if (endpointNode) {
                        explicitCornerTetherPos.push(outerPt.x, outerPt.y, outerPt.z);
                        explicitCornerTetherPos.push(endpointNode.position.x, endpointNode.position.y, endpointNode.position.z);
                    }
                });

                for (let i = 0; i < sys.nodesA.length; i++) {
                    const nodeA = sys.nodesA[i];
                    const inverseIndex = (sys.nodesB.length - 1 - ((i * currentPrimeSeed) % sys.nodesB.length));
                    const nodeB = sys.nodesB[inverseIndex];
                    closedDualbellTetherPos.push(nodeA.position.x, nodeA.position.y, nodeA.position.z);

                    closedDualbellTetherPos.push(nodeB.position.x, nodeB.position.y, nodeB.position.z);
                }

                for (let i = 0; i < sys.nodesA.length; i++) {
                    const nodeA = sys.nodesA[i];
                    const nodeB = sys.nodesB[i];
                    if (i % 2 === 0) {
                        horizontalMirrorPos.push(nodeA.position.x, nodeA.position.y, nodeA.position.z);
                        horizontalMirrorPos.push(nodeB.position.x, nodeB.position.y, nodeB.position.z);
                    } else {
                        const twistOffsetIndex = (i + currentCoprimeQ) % sys.nodesB.length;
                        horizontalMirrorPos.push(nodeA.position.x, nodeA.position.y, nodeA.position.z);
                        horizontalMirrorPos.push(sys.nodesB[twistOffsetIndex].position.x, sys.nodesB[twistOffsetIndex].position.y, sys.nodesB[twistOffsetIndex].position.z);
                    }
                }

                sys.quadTethers.geometry.setAttribute('position', new THREE.Float32BufferAttribute(quadStabilizerPos, 3));
                sys.quadTethers.geometry.attributes.position.needsUpdate = true;


                sys.horizonCornerTethers.geometry.setAttribute('position', new THREE.Float32BufferAttribute(explicitCornerTetherPos, 3));
                sys.horizonCornerTethers.geometry.attributes.position.needsUpdate = true;

                sys.dualbellInterlinks.geometry.setAttribute('position', new THREE.Float32BufferAttribute(closedDualbellTetherPos, 3));
                sys.dualbellInterlinks.geometry.attributes.position.needsUpdate = true;

                sys.horizontalMirrorTethers.geometry.setAttribute('position', new THREE.Float32BufferAttribute(horizontalMirrorPos, 3));
                sys.horizontalMirrorTethers.geometry.attributes.position.needsUpdate = true;
            }

            function animate() {
                requestAnimationFrame(animate);

                executionTime += 16.666;
                const secondsElapsed = executionTime * 0.001;
                
                updateClockString(executionTime);
                updateTelemetryReadouts(secondsElapsed);
                processFlightControls();


                const positionAttribute = realGridGeo.attributes.position;
                const transformationDepthVal = parseInt(inputDepth.value);
                
                for (let i = 0; i < positionAttribute.count; i++) {
                    const x = realOrigPos.getX(i);
                    const y = realOrigPos.getY(i);
                    const waveZ = Math.sin(x * 0.15 + secondsElapsed * 1.5) * Math.cos(y * 0.15 + secondsElapsed * 1.2) * (transformationDepthVal * 0.6);
                    positionAttribute.setZ(i, waveZ);
                }
                positionAttribute.needsUpdate = true;

                adjacentFieldGroup.rotation.y = -secondsElapsed * 0.15;
                if (adjacentLatticeMesh) {
                    adjacentLatticeMesh.rotation.z = Math.sin(secondsElapsed * 0.4) * (currentMobiusFactor * 0.05);
                }

                sierpinskiGroup.rotation.y = secondsElapsed * 0.25;
                sierpinskiGroup.rotation.x = secondsElapsed * 0.08;


                anchorMarkers.forEach((marker) => {
                    const waveMod = Math.sin(secondsElapsed * 2.2 + marker.index) * 0.25;
                    marker.mesh.scale.set(1.0 + waveMod, 1.0 + waveMod, 1.0 + waveMod);
                    
                    const markerHue = chaosOverflow(secondsElapsed, marker.index, (secondsElapsed * 0.06) % 1.0);
                    marker.mesh.material.color.setHSL(markerHue, 1.0, marker.type === 'FACE' ? 0.55 : 0.35);
                });

                const independentColorWaveFront = secondsElapsed * 1.3;
                updateSubsystem(alphaSystem, secondsElapsed, independentColorWaveFront);
                updateSubsystem(omegaSystem, secondsElapsed, independentColorWaveFront);

                const bridgeCoordinates = [];
                const totalLanesCount = alphaSystem.rivetedLanes.length;

                for (let l = 0; l < totalLanesCount - 2; l++) {
                    const laneA = alphaSystem.rivetedLanes[l];
                    const rivetA = laneA.rivets[laneA.rivets.length - 1].mesh; 
                    const worldPosA = new THREE.Vector3();

                    rivetA.getWorldPosition(worldPosA);

                    const targetIndexA = (l * currentPrimeSeed) % combinedAnchorsPool.length;
                    const localAnchorPosA = combinedAnchorsPool[targetIndexA].pos.clone();
                    sierpinskiGroup.localToWorld(localAnchorPosA);

                    bridgeCoordinates.push(worldPosA.x, worldPosA.y, worldPosA.z);
                    bridgeCoordinates.push(localAnchorPosA.x, localAnchorPosA.y, localAnchorPosA.z);

                    const laneB = omegaSystem.rivetedLanes[l];
                    const rivetB = laneB.rivets[0].mesh; 
                    const worldPosB = new THREE.Vector3();
                    rivetB.getWorldPosition(worldPosB);

                    const targetIndexB = (l * currentPrimeSeed + currentCoprimeQ) % combinedAnchorsPool.length;
                    const localAnchorPosB = combinedAnchorsPool[targetIndexB].pos.clone();
                    sierpinskiGroup.localToWorld(localAnchorPosB);

                    bridgeCoordinates.push(worldPosB.x, worldPosB.y, worldPosB.z);

                    bridgeCoordinates.push(localAnchorPosB.x, localAnchorPosB.y, localAnchorPosB.z);
                }

                const polarSortedAnchors = [...combinedAnchorsPool].sort((a, b) => b.pos.y - a.pos.y);
                const northPolarEdge = polarSortedAnchors[0].pos;
                const southPolarEdge = polarSortedAnchors[polarSortedAnchors.length - 1].pos;

                const alphaPolar1 = alphaSystem.rivetedLanes[10].rivets[alphaSystem.rivetedLanes[10].rivets.length - 1].mesh;
                const worldAlphaPolar1 = new THREE.Vector3();
                alphaPolar1.getWorldPosition(worldAlphaPolar1);
                const globalNorthA = northPolarEdge.clone();
                sierpinskiGroup.localToWorld(globalNorthA);
                bridgeCoordinates.push(worldAlphaPolar1.x, worldAlphaPolar1.y, worldAlphaPolar1.z);
                bridgeCoordinates.push(globalNorthA.x, globalNorthA.y, globalNorthA.z);

                const alphaPolar2 = alphaSystem.rivetedLanes[11].rivets[alphaSystem.rivetedLanes[11].rivets.length - 1].mesh;
                const worldAlphaPolar2 = new THREE.Vector3();
                alphaPolar2.getWorldPosition(worldAlphaPolar2);
                const globalSouthA = southPolarEdge.clone();

                sierpinskiGroup.localToWorld(globalSouthA);
                bridgeCoordinates.push(worldAlphaPolar2.x, worldAlphaPolar2.y, worldAlphaPolar2.z);
                bridgeCoordinates.push(globalSouthA.x, globalSouthA.y, globalSouthA.z);

                const omegaPolar1 = omegaSystem.rivetedLanes[10].rivets[0].mesh;
                const worldOmegaPolar1 = new THREE.Vector3();
                omegaPolar1.getWorldPosition(worldOmegaPolar1);
                const globalNorthB = northPolarEdge.clone();
                sierpinskiGroup.localToWorld(globalNorthB);
                bridgeCoordinates.push(worldOmegaPolar1.x, worldOmegaPolar1.y, worldOmegaPolar1.z);
                bridgeCoordinates.push(globalNorthB.x, globalNorthB.y, globalNorthB.z);

                const omegaPolar2 = omegaSystem.rivetedLanes[11].rivets[0].mesh;
                const worldOmegaPolar2 = new THREE.Vector3();
                omegaPolar2.getWorldPosition(worldOmegaPolar2);
                const globalSouthB = southPolarEdge.clone();
                sierpinskiGroup.localToWorld(globalSouthB);
                bridgeCoordinates.push(worldOmegaPolar2.x, worldOmegaPolar2.y, worldOmegaPolar2.z);
                bridgeCoordinates.push(globalSouthB.x, globalSouthB.y, globalSouthB.z);


                integrationBridgeLines.geometry.setAttribute('position', new THREE.Float32BufferAttribute(bridgeCoordinates, 3));
                integrationBridgeLines.geometry.attributes.position.needsUpdate = true;

                renderer.render(scene, camera);
            }

            animate();
        };
